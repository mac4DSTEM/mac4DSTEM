// ---- footprint guard: kill above 2.5 GB phys_footprint
func footprint() -> UInt64 {
    var info = task_vm_info_data_t()
    var count = mach_msg_type_number_t(MemoryLayout<task_vm_info_data_t>.size / MemoryLayout<integer_t>.size)
    let kr = withUnsafeMutablePointer(to: &info) {
        $0.withMemoryRebound(to: integer_t.self, capacity: Int(count)) { task_info(mach_task_self_, task_flavor_t(TASK_VM_INFO), $0, &count) }
    }
    return kr == KERN_SUCCESS ? info.phys_footprint : 0
}
final class Guard: @unchecked Sendable {
    var peak: UInt64 = 0
    func start() {
        let t = Thread { [self] in
            while true {
                let f = footprint(); if f > peak { peak = f }
                if f > 2_500_000_000 { FileHandle.standardError.write(Data("GUARD: footprint \(f) > 2.5 GB, exiting\n".utf8)); exit(99) }
                usleep(200_000)
            }
        }
        t.start()
    }
}

struct Profile {
    var rOut: Int; var D: Double; var Dfull: Double; var rRing: Int; var rRingFull: Int; var prof: [Double]
}
func profile(dp: [Float], qy: Int, qx: Int, x0: Float, y0: Float, rEst: Float) -> Profile {
    let nb = Int(Double(qx * qx + qy * qy).squareRoot()) + 2
    var sum = [Double](repeating: 0, count: nb), cnt = [Double](repeating: 0, count: nb)
    for y in 0..<qy { for x in 0..<qx {
        let v = dp[y * qx + x]; guard v.isFinite else { continue }
        let r = Int(hypot(Double(y) - Double(y0), Double(x) - Double(x0)))
        sum[r] += Double(v); cnt[r] += 1
    } }
    let prof = (0..<nb).map { cnt[$0] > 0 ? sum[$0] / cnt[$0] : 0 }
    let m = prof.max() ?? 1
    var rOut = 0
    for i in 0..<nb where prof[i] >= 0.2 * m { rOut = i + 1 }
    func ring(limit: Int) -> (Int, Double) {
        var last = 0
        for i in 1..<(min(limit, nb - 1)) where prof[i] > prof[i - 1] && prof[i] >= prof[i + 1] && prof[i] >= 0.2 * m { last = i }
        if last <= 1 { return (last, 1) }
        return (last, (1..<last).map { prof[$0] }.min()! / m)
    }
    let a = ring(limit: Int(2 * rEst) + 1), b = ring(limit: nb - 1)
    return Profile(rOut: rOut, D: a.1, Dfull: b.1, rRing: a.0, rRingFull: b.0, prof: Array(prof.prefix(40)).map { ($0 / m * 1000).rounded() / 1000 })
}

// ---- scoring
struct Score { var pred = 0; var m2 = 0; var m3 = 0; var beam2 = false; var beamBright = false }
func score(peaks: [BraggPeak], labels: [[Double]], swap: Bool = false) -> Score {
    var s = Score(); s.pred = peaks.count
    let pts = peaks.map { swap ? (Double($0.x), Double($0.y)) : (Double($0.y), Double($0.x)) } // (row, col)
    func match(_ tol: Double) -> (Int, Set<Int>) {
        var pairs: [(Double, Int, Int)] = []
        for (i, p) in pts.enumerated() { for (j, l) in labels.enumerated() {
            let d = hypot(p.0 - l[0], p.1 - l[1]); if d <= tol { pairs.append((d, i, j)) } } }
        pairs.sort { $0.0 < $1.0 }
        var up = Set<Int>(), ul = Set<Int>()
        for (_, i, j) in pairs where !up.contains(i) && !ul.contains(j) { up.insert(i); ul.insert(j) }
        return (up.count, ul)
    }
    let (m2, l2) = match(2); s.m2 = m2; s.m3 = match(3).0
    s.beam2 = l2.contains(0)
    if let b = peaks.enumerated().max(by: { $0.element.intensity < $1.element.intensity }) {
        let p = pts[b.offset]; s.beamBright = hypot(p.0 - labels[0][0], p.1 - labels[0][1]) <= 2
    }
    return s
}

import json, sys, collections

def agg(rs, key="m2"):
    nl = sum(r["nlab"] for r in rs); npred = sum(r["pred"] for r in rs); m = sum(r[key] for r in rs)
    return dict(rec=m / nl, prec=m / npred if npred else 0, npos=len(rs), pred=npred / len(rs),
                cap=sum(1 for r in rs if r["pred"] >= 70) / len(rs), beam=sum(1 for r in rs if r["beamBright"]) / len(rs),
                beam2=sum(1 for r in rs if r["beam2"]) / len(rs))
def halves(name):
    rs = by[name]
    return {"sel": agg([r for r in rs if r["pos"] % 2 == 0]), "conf": agg([r for r in rs if r["pos"] % 2 == 1]), "all": agg(rs),
            "all3": agg(rs, "m3")}
